#[cfg(test)]
mod tests {
use super::{models::*,json_codecs::*};
use serde_json::json;
#[test] fn aliases_presence_and_typed_false_default() {
 let mut model=read_Record_json(r#"{"old_label":"hello","enabled":false,"items":[]}"#).unwrap();assert!(matches!(model.amount,Presence::Absent));assert_eq!(serde_json::to_value(&model).unwrap()["enabled"],false);
 assert_eq!(serde_json::from_str::<serde_json::Value>(&write_Record_json(&model).unwrap()).unwrap(),json!({"label":"hello"}));
 model.enabled=Presence::Absent;let absent=write_Record_json(&model).unwrap();assert!(!absent.contains("enabled"));
 let null=read_Record_json(r#"{"label":null,"amount":null}"#).unwrap();assert!(matches!(null.title,Presence::Null));assert!(matches!(null.amount,Presence::Null));assert_eq!(serde_json::from_str::<serde_json::Value>(&write_Record_json(&null).unwrap()).unwrap(),json!({"label":null,"amount":null}));
}
#[test] fn exact_boundaries_use_merged_scalar_validation() {
 let wire=r#"{"amount":"123456789012345678901234567890.00100","count":"9007199254740993123456789"}"#;let model=read_Record_json(wire).unwrap();assert_eq!(serde_json::from_str::<serde_json::Value>(&write_Record_json(&model).unwrap()).unwrap(),serde_json::from_str::<serde_json::Value>(wire).unwrap());
 assert!(read_Record_json(r#"{"count":9007199254740993}"#).is_err());assert!(read_Record_json(r#"{"amount":0.1}"#).is_err());assert!(read_Record_json(r#"{"amount":"1e100"}"#).is_err());
}
#[test] fn nested_models_and_discriminators_use_wire_aliases() {
 let model=read_Record_json(r#"{"child":{"old_name":"nested"},"selected":{"kind":"item","wire_name":"selected"}}"#).unwrap();let output=serde_json::from_str::<serde_json::Value>(&write_Record_json(&model).unwrap()).unwrap();assert_eq!(output,json!({"child":{"wire_name":"nested"},"selected":{"kind":"item","wire_name":"selected"}}));
 let err=read_Record_json(r#"{"selected":{"kind":"unknown"}}"#).unwrap_err();assert_eq!(err.reason,super::contracts::ErrorReason::UnknownDiscriminator);
 assert_eq!(read_Record_json(r#"{"selected":{}}"#).unwrap_err().reason,super::contracts::ErrorReason::UnknownDiscriminator);
}
#[test] fn root_keyed_collection_preserves_remaining_wire_mappings() {
 let wire=r#"{"first":{"wire_name":"a","price":"2.00"},"second":{"wire_name":"b"}}"#;let model=read_Root_json(wire).unwrap();let output=write_Root_json(&model).unwrap();assert_eq!(serde_json::from_str::<serde_json::Value>(&output).unwrap(),serde_json::from_str::<serde_json::Value>(wire).unwrap());
}
#[test] fn child_keyed_collection_preserves_nested_paths() {
 let wire=r#"{"records":{"a":{"path":{"name":"alpha"},"price":"3.100"},"b":{"path":{"name":"beta"}}}}"#;let model=read_Children_json(wire).unwrap();let output=write_Children_json(&model).unwrap();assert_eq!(serde_json::from_str::<serde_json::Value>(&output).unwrap(),serde_json::from_str::<serde_json::Value>(wire).unwrap());
}
#[test] fn malformed_unknown_duplicate_and_alias_conflicts_reject() {
 for wire in ["{",r#"{"unknown":1}"#,r#"{"label":"a","label":"b"}"#,r#"{"label":"a","old_label":"b"}"#,r#"{"enabled":"false"}"#] {let error=read_Record_json(wire).unwrap_err();assert_eq!(error.reason,super::contracts::ErrorReason::InvalidDocument);assert_eq!(error.format.as_deref(),Some("json"));}
}

#[test] fn mixed_content_tokens_survive_json_mapping() {
 let wire=r#"{"label":"title","__lml_content":[{"kind":"text","literal":"before"},{"kind":"element","attribute":"title","index":0},{"kind":"cdata","literal":"after"}]}"#;
 let model=read_Record_json(wire).unwrap();let output=write_Record_json(&model).unwrap();assert_eq!(serde_json::from_str::<serde_json::Value>(&output).unwrap(),serde_json::from_str::<serde_json::Value>(wire).unwrap());
}
}
